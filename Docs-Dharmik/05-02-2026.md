// --- Frontend Pagination & Filtering Logic ---
let allRows = [];
let filteredRows = [];
let currentPage = 1;
const itemsPerPage = 10;

document.addEventListener('DOMContentLoaded', function() {
    const rows = document.querySelectorAll('.appointment-row');
    rows.forEach(row => allRows.push(row));
    applyFilters();
});

function applyFilters() {
    const searchText = document.getElementById('searchInput').value.toLowerCase();
    const dateValue = document.getElementById('dateFilter').value;

    filteredRows = allRows.filter(row => {
        const name = row.getAttribute('data-name');
        const date = row.getAttribute('data-date'); 
        const matchesName = name.includes(searchText);
        const matchesDate = dateValue ? date === dateValue : true;
        return matchesName && matchesDate;
    });

    currentPage = 1;
    renderTable();
}

## 2. Backend Changes

### A. Appointments View (`appointments/views.py`)

**Function: `hospital_appointments_view`**
*Logic Added: Fetch phone number with fallback to Name lookup*

```python
# Apply Phone Logic
appointments_list = []
for appt in appointments:
    appt.patient_phone = ""
    # 1. Try via Email
    if appt.patient_email:
        try:
            patient_obj = Patient.objects.filter(user__email=appt.patient_email).first()
            if patient_obj:
                appt.patient_phone = patient_obj.phone
        except Exception:
            pass
    
    # 2. Fallback: Try via Name
    if not appt.patient_phone and appt.patient_name:
        try:
            patient_obj = Patient.objects.filter(name__iexact=appt.patient_name).first()
            if patient_obj:
                    appt.patient_phone = patient_obj.phone
            else:
                    patient_obj = Patient.objects.filter(user__username__iexact=appt.patient_name).first()
                    if patient_obj:
                        appt.patient_phone = patient_obj.phone
        except Exception:
            pass
    appointments_list.append(appt)
appointments = appointments_list # Update context variable
```
